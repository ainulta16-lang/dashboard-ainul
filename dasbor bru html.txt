/* ==================================

var passwordBenar = "ainul16";


/* ==================================
   FUNGSI MASUK
================================== */

function masuk(){

    var namaInput =
        document.getElementById("nama");

    var passwordInput =
        document.getElementById("password");

    var error =
        document.getElementById("error");


    var nama =
        namaInput.value.trim();

    var password =
        passwordInput.value;


    /* CEK NAMA */

    if(nama === ""){

        error.innerHTML =
            "⚠️ Masukkan nama terlebih dahulu.";

        error.style.display =
            "block";

        namaInput.focus();

        return;

    }


    /* CEK PASSWORD */

    if(password === ""){

        error.innerHTML =
            "⚠️ Masukkan kata sandi.";

        error.style.display =
            "block";

        passwordInput.focus();

        return;

    }


    /* PASSWORD BENAR */

    if(password === passwordBenar){

        document
        .getElementById("namaTampil")
        .innerHTML = nama;


        document
        .getElementById("login")
        .style.display = "none";


        document
        .getElementById("dashboard")
        .style.display = "block";


        error.style.display =
            "none";

    }


    /* PASSWORD SALAH */

    else{

        error.innerHTML =
            "❌ Kata sandi salah. Coba lagi.";

        error.style.display =
            "block";

        passwordInput.focus();

    }

}


/* ==================================
   ENTER UNTUK MASUK
================================== */

document
.getElementById("password")
.addEventListener("keydown",function(event){

    if(event.key === "Enter"){

        masuk();

    }

});
